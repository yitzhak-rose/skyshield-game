file (GLOB_RECURSE GAME_RESOURCE_FILES CONFIGURE_DEPENDS
    LIST_DIRECTORIES false
    RELATIVE ${CMAKE_CURRENT_LIST_DIR}
    data/*
    fonts/*
    images/*
    sounds/*
)

foreach (GAME_RESOURCE_FILE ${GAME_RESOURCE_FILES})
    cmake_path(GET GAME_RESOURCE_FILE PARENT_PATH GAME_RESOURCE_DIR)
    if (NOT "${GAME_RESOURCE_DIR}" STREQUAL "")
        file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/${GAME_RESOURCE_DIR})
    endif ()
    configure_file (${GAME_RESOURCE_FILE} ${CMAKE_BINARY_DIR}/${GAME_RESOURCE_FILE} COPYONLY)
endforeach ()
